Feature-4 : CREATE FUNCTION SAVE (ETU003301)
DemandeurService :
    - Demandeur save(Demandeur demandeur) {
        envoi d'exception si demandeur null
        appel de la fonction save de DemandeurRepository
        retourner demandeur
    }
PasseportService : 
    - Passeport save(Passeport passeport) {
        envoi d'exception si passeport null
        appel de la fonction save de PasseportRepository
        retourner passeport
    }
VisaTransformableService : 
    - VisaTransformable save(VisaTransformable visaTransformable) {
        envoi d'exception si visaTransformable null
        appel de la fonction save de VisaTransformableRepository
        retourner visaTransformable
    }
DemandeService : 
    - Demande save(Demande demande) {
        envoi d'exception si demande null
        appel de la fonction save de DemandeRepository
        retourner demande
    }
    - String saveNouveuTitre(Demandeur demandeur, Passeport passeport, VisaTransformable visaTransformable, List<Document> documents, Long idCategorieDemande) {
        appel de la fonction save de DemandeurService
        si la fonction retourne null -> exception = "Echec de l'enregistrement du demandeur"
        appel de la fonction save de PasseportService
        si la fonction retourne null -> exception = "Echec de l'enregistrement du passeport"
        declarer la variable Demande demande
        utilisation de setters de demande
        appel de la fonction save de DemandeService
        si la fonction retourne null -> exception = "Echec de l'enregistrement de la demande"
        appel de la fonction save de VisaTransformableService
        si la fonction retourne null -> exception = "Echec de l'enregistrement du visa transformable"
        appel de la fonction save de HistoriqueDemandeService
        si la fonction retourne null -> exception = "Echec de l'enregistrement de la statut de demande"
        retourner un message de reussite 
    }
HistoriqueDemandeService :
    - HistoriqueDemande save(HistoriqueDemande historiqueDemande) {
        envoi d'exception si historiqueDemande null
        appel de la fonction save de historiqueDemande
        retourner historiqueDemande
    }
